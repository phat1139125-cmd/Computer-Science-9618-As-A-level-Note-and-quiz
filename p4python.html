SECTION 01
Searching Algorithms
LINEAR SEARCH · BINARY SEARCH
Linear Search
def linear_search(arr, target): for i in range(len(arr)): if arr[i] == target: return i # return index if found return -1 # return -1 if not found # With boolean flag (use when action needed if NOT found) def linear_search_flag(arr, target): Found = False Index = -1 for i in range(len(arr)): if arr[i] == target: Found = True Index = i if not Found: print("Not found") return Index
Binary Search
def binary_search(arr, target): First = 0 Last = len(arr) - 1 while First <= Last: MidValue = (First + Last) // 2 if arr[MidValue] == target: return MidValue # found elif arr[MidValue] > target: Last = MidValue - 1 # discard upper half else: First = MidValue + 1 # discard lower half return -1 # not found # ⚠️ Binary search requires SORTED array # First = first index of section being searched # Last = last index of section being searched # MidValue = index of the middle element
Key Notes
  • Linear: no precondition, O(n)
  • Binary: array MUST be sorted, O(log n)
  • Return -1 = not found convention
  • Binary: if value at MidValue > target → Last = MidValue-1
  • Binary: if value at MidValue < target → First = MidValue+1
SECTION 02
Sorting Algorithms
BUBBLE SORT · INSERTION SORT
Bubble Sort
def bubble_sort(arr): n = len(arr) for i in range(n): for j in range(n - 1 - i): # -i leaves sorted end out if arr[j] > arr[j + 1]: # change > to < for descending # Swap arr[j], arr[j + 1] = arr[j + 1], arr[j] return arr # ⚡ Outer loop: runs n times # ⚡ Inner loop: n-1-i because last i elements already sorted # ⚡ Change > to < for DESCENDING order
Insertion Sort
def insertion_sort(arr): for i in range(1, len(arr)): Current = arr[i] # element being inserted j = i - 1 while j >= 0 and arr[j] > Current: arr[j + 1] = arr[j] # shift element right j -= 1 arr[j + 1] = Current # insert in correct position return arr # ⚡ Outer loop: moves through each element from index 1 # ⚡ Current: stores element being compared against all before it # ⚡ Inner while: shifts larger elements right until correct slot found # ⚡ More efficient than bubble sort for nearly-sorted data
SECTION 03
File Handling
READ · WRITE · APPEND · EXCEPTION HANDLING
# Reading from a file — always use exception handling try: with open("filename.txt", "r") as f: for line in f: line = line.strip() # removes \n from end print(line) except FileNotFoundError: print("File not found") # Reading into an array data = [] try: with open("filename.txt", "r") as f: for line in f: data.append(line.strip()) # .strip() removes \n except FileNotFoundError: print("File not found") # Writing to a file (OVERWRITES existing content) with open("filename.txt", "w") as f: f.write("data\n") # \n = new line # Appending to a file (PRESERVES existing content) with open("filename.txt", "a") as f: f.write("new data\n")
"r" / "w" / "a" modes
  • "r" = READ — file must exist
  • "w" = WRITE — OVERWRITES all existing data
  • "a" = APPEND — adds to end, preserves existing
  • Always use .strip() to remove \n when reading
  • Always use exception handling for file operations

Stack ADT
INITIALISE · PUSH · POP · IS_EMPTY · IS_FULL
# Stack using array + TopPointer MAX_SIZE = 10 Stack = [None] * MAX_SIZE TopPointer = -1 # -1 = empty stack # PUSH — add to stack def Push(item): global TopPointer if TopPointer < MAX_SIZE - 1: # check not full TopPointer += 1 Stack[TopPointer] = item else: print("Stack is full") # POP — remove from stack def Pop(): global TopPointer if TopPointer >= 0: # check not empty item = Stack[TopPointer] TopPointer -= 1 return item else: print("Stack is empty") return None # IS_EMPTY and IS_FULL def IsEmpty(): return TopPointer == -1 def IsFull(): return TopPointer == MAX_SIZE - 1 # ⚡ TopPointer starts at -1 (empty) # ⚡ PUSH: check not full FIRST, then increment pointer, then store # ⚡ POP: check not empty FIRST, store value, then decrement pointer
SECTION 07
Queue ADT
LINEAR QUEUE · CIRCULAR QUEUE · ENQUEUE · DEQUEUE
# Circular Queue MAX_SIZE = 10 Queue = [None] * MAX_SIZE HeadPointer = 0 # points to first element TailPointer = 0 # points to next free slot Count = 0 # tracks number of items # ENQUEUE — add to queue def Enqueue(item): global TailPointer, Count if Count < MAX_SIZE: # check not full Queue[TailPointer] = item TailPointer = (TailPointer + 1) % MAX_SIZE # circular wrap Count += 1 else: print("Queue is full") # DEQUEUE — remove from queue def Dequeue(): global HeadPointer, Count if Count > 0: # check not empty item = Queue[HeadPointer] HeadPointer = (HeadPointer + 1) % MAX_SIZE # circular wrap Count -= 1 return item else: print("Queue is empty") return None # ⚡ (pointer + 1) % MAX_SIZE = circular wrap-around # ⚡ Count variable tracks items — avoids Head==Tail ambiguity # ⚡ Head and Tail both start at 0 for circular queue

Linked List ADT
INITIALISE · INSERT · DELETE · TRAVERSE
# Linked list using two arrays: Data[] and Pointer[] MAX_SIZE = 10 Data = [-1] * MAX_SIZE # -1 = empty node Pointer = [i + 1 for i in range(MAX_SIZE)] # free list Pointer[MAX_SIZE - 1] = -1 # last node has null pointer FirstNode = -1 # -1 = list empty FirstEmpty = 0 # first empty node = index 0 # INSERT at beginning def Insert(value): global FirstNode, FirstEmpty if FirstEmpty == -1: # check list not full print("List is full") return NextEmpty = Pointer[FirstEmpty] # save next empty Data[FirstEmpty] = value # store value Pointer[FirstEmpty] = FirstNode # point to old first FirstNode = FirstEmpty # update first node FirstEmpty = NextEmpty # update free pointer # DELETE a value def Delete(value): global FirstNode, FirstEmpty if FirstNode == -1: return # empty list if Data[FirstNode] == value: # delete first node Temp = FirstNode FirstNode = Pointer[FirstNode] Pointer[Temp] = FirstEmpty Data[Temp] = -1 FirstEmpty = Temp return PreviousNode = FirstNode # search rest of list CurrentNode = Pointer[FirstNode] while CurrentNode != -1: if Data[CurrentNode] == value: Pointer[PreviousNode] = Pointer[CurrentNode] Pointer[CurrentNode] = FirstEmpty Data[CurrentNode] = -1 FirstEmpty = CurrentNode return PreviousNode = CurrentNode CurrentNode = Pointer[CurrentNode] # TRAVERSE (output all) def Traverse(): Pointer_val = FirstNode while Pointer_val != -1: print(Data[Pointer_val]) Pointer_val = Pointer[Pointer_val] # ⚡ -1 = null pointer (no node) # ⚡ Data[i] = -1 means node i is empty # ⚡ FirstNode points to head; FirstEmpty to first free node

# Recursive Factorial def factorial(n): if n == 0 or n == 1: # base case return 1 return n * factorial(n - 1) # general case # Recursive Fibonacci def fibonacci(n): if n <= 1: # base case return n return fibonacci(n - 1) + fibonacci(n - 2) # general case # Recursive Binary Search def rec_binary_search(arr, target, first, last): if first > last: # base case — not found return -1 mid = (first + last) // 2 if arr[mid] == target: # base case — found return mid elif arr[mid] > target: return rec_binary_search(arr, target, first, mid - 1) else: return rec_binary_search(arr, target, mid + 1, last) # Call: rec_binary_search(arr, target, 0, len(arr)-1) # ⚡ Every recursive function MUST have: # 1. Base case (stopping condition) # 2. General case (calls itself, moves toward base case)
